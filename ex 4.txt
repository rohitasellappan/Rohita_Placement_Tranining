import java.util.*;

public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		String str = sc.next();
		int fir =0,last=0;
		while(fir < last) {
		    if(str.charAt(fir) != str.charAt(last)) {
		        System.out.print("Not");
		        return;
		    }
		    fir++;
		    last--;
		}
		System.out.print("Palindrome");
	}
}
